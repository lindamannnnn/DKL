### 2. 知识点讲解

#### 2.1 BFS 最短路的原理

BFS 的层次性保证了最优性：

> 当 BFS 从起点出发扩展时，第 k 层的所有节点到起点的距离恰好是 k。
>
> 因此，第一次到达终点时，当前层数 k 就是最短路距离。

用 `dis` 数组替代 `vis` 数组，同时记录距离：

```cpp
int dis[105][105];
memset(dis, -1, sizeof(dis));  // -1 表示未访问
dis[sx][sy] = 0;               // 起点距离为 0

// 扩展时：
dis[nx][ny] = dis[x][y] + 1;  // 更新距离
```

#### 2.2 前驱数组：记录路径

```cpp
pair<int,int> prev[105][105];  // prev[x][y] = 到达(x,y)的前驱坐标
// 初始化
memset(prev, -1, sizeof(prev));

// BFS 扩展时记录前驱：
if (dis[nx][ny] == -1) {
    dis[nx][ny] = dis[x][y] + 1;
    prev[nx][ny] = {x, y};   // 记录前驱
    q.push({nx, ny});
}
```

#### 2.3 路径重建：从终点反向追踪

```cpp
void printPath(int ex, int ey) {
    vector<pair<int,int>> path;
    int cx=ex, cy=ey;
    while (cx != -1) {
        path.push_back({cx, cy});
        auto [px, py] = prev[cx][cy];
        cx = px; cy = py;
    }
    reverse(path.begin(), path.end());
    for (auto [x,y] : path) cout << "(" << x << "," << y << ") ";
}
```

#### 2.4 完整代码：BFS 最短路 + 路径输出

```cpp
#include <bits/stdc++.h>
using namespace std;
int n,m,a[105][105],dis[105][105];
pair<int,int> prev_[105][105];
int dx[]={0,-1,1,0,0}, dy[]={0,0,0,-1,1};

void bfs(int sx, int sy) {
    memset(dis,-1,sizeof(dis));
    memset(prev_,-1,sizeof(prev_));
    queue<pair<int,int>> q;
    dis[sx][sy]=0; q.push({sx,sy});
    while(!q.empty()) {
        auto [x,y]=q.front(); q.pop();
        for(int i=1;i<=4;i++) {
            int nx=x+dx[i], ny=y+dy[i];
            if(nx>=1&&nx<=n&&ny>=1&&ny<=m&&a[nx][ny]==0&&dis[nx][ny]==-1) {
                dis[nx][ny]=dis[x][y]+1;
                prev_[nx][ny]={x,y};
                q.push({nx,ny});
            }
        }
    }
}

int main() {
    cin>>n>>m;
    for(int i=1;i<=n;i++) for(int j=1;j<=m;j++) cin>>a[i][j];
    bfs(1,1);
    cout<<"最短步数：" << dis[n][m] << "\n";
    // 输出路径
    vector<pair<int,int>> path;
    int cx=n, cy=m;
    while(cx!=-1) {
        path.push_back({cx,cy});
        auto [px,py]=prev_[cx][cy];
        cx=px; cy=py;
    }
    reverse(path.begin(),path.end());
    for(auto [x,y]:path) cout<<"("<<x<<","<<y<<") ";
    return 0;
}
```

#### 2.5 统计最短路条数

```cpp
int cnt[105][105];  // 到达 (x,y) 的最短路条数
cnt[sx][sy]=1;

// BFS 扩展时：
if(dis[nx][ny]==-1) {
    dis[nx][ny]=dis[x][y]+1;
    cnt[nx][ny]=cnt[x][y];   // 继承前驱的路径数
    q.push({nx,ny});
} else if(dis[nx][ny]==dis[x][y]+1) {
    cnt[nx][ny]+=cnt[x][y];  // 发现另一条等长最短路
}
```

### 3. GESP 真题演练 ⚡

**抢答题 1（选择题）：**

> BFS 求最短路时，当终点第一次被弹出队列时，其 `dis` 值（ ）
>
> A. 可能不是最短距离  B. 一定是最短距离  C. 是所有路径长度的平均值  D. 取决于起点
>
> **答案：B**
>
> 解析：BFS 层次保证——第k层的节点距起点恰好为k，第一次到达即是最短。

**判断题 2：**

> BFS 在有权图（边有不同权重）中也能保证找到最短路。（ ）
>
> **答案：✗**
>
> 解析：BFS 只在无权图（或边权为1）时保证最短路。有权图需要用 Dijkstra 算法。

**抢答题 3（选择题）：**

> 前驱数组 `prev[nx][ny]={x,y}` 在 BFS 中的作用是（ ）
>
> A. 记录所有可能的路径  B. 记录最短路径上每个节点的来源，用于路径重建
>
> C. 统计最短路的条数  D. 优化 BFS 速度
>
> **答案：B**

### 4. 进阶扩展

**BFS 最短路总结：**

> 黑板推演：BFS 最短路的三个核心组件
>
> 1. `dis[x][y]`：起点到(x,y)的最短距离，初始化为-1
>
> 2. `prev[x][y]`：最短路上(x,y)的前驱坐标，用于路径重建
>
> 3. `cnt[x][y]`：到达(x,y)的最短路条数，用于计数

```cpp
// 完整最短路模板
memset(dis,-1,sizeof(dis)); dis[sx][sy]=0;
prev_[sx][sy]={-1,-1};
cnt[sx][sy]=1;
queue<pair<int,int>> q; q.push({sx,sy});
while(!q.empty()) {
    auto [x,y]=q.front(); q.pop();
    for(int i=0;i<4;i++) {
        int nx=x+dx[i], ny=y+dy[i];
        if(合法 && dis[nx][ny]==-1) {
            dis[nx][ny]=dis[x][y]+1;
            prev_[nx][ny]={x,y};
            cnt[nx][ny]=cnt[x][y];
            q.push({nx,ny});
        } else if(合法 && dis[nx][ny]==dis[x][y]+1) {
            cnt[nx][ny]+=cnt[x][y];
        }
    }
}
```

### 练习 3（综合）：最短路条数

n×m迷宫（0可走1是墙），统计从(1,1)到(n,m)的最短路条数。

**输入格式：** n m 及迷宫（1≤n,m≤20）

**输出格式：** 最短路条数（答案对1e9+7取模）

**样例输入：**
```
3 3
0 0 0
0 0 0
0 0 0
```
**样例输出：** `6`

**题解代码：**

```cpp
#include <bits/stdc++.h>
using namespace std;
const int MOD=1e9+7;
int n,m,a[25][25],dis[25][25];
long long cnt[25][25];
int dx[]={-1,1,0,0}, dy[]={0,0,-1,1};
int main() {
    cin>>n>>m;
    for(int i=1;i<=n;i++) for(int j=1;j<=m;j++) cin>>a[i][j];
    memset(dis,-1,sizeof(dis));
    dis[1][1]=0; cnt[1][1]=1;
    queue<pair<int,int>> q; q.push({1,1});
    while(!q.empty()) {
        auto [x,y]=q.front(); q.pop();
        for(int i=0;i<4;i++) {
            int nx=x+dx[i], ny=y+dy[i];
            if(nx>=1&&nx<=n&&ny>=1&&ny<=m&&a[nx][ny]==0) {
                if(dis[nx][ny]==-1) {
                    dis[nx][ny]=dis[x][y]+1;
                    cnt[nx][ny]=cnt[x][y];
                    q.push({nx,ny});
                } else if(dis[nx][ny]==dis[x][y]+1) {
                    cnt[nx][ny]=(cnt[nx][ny]+cnt[x][y])%MOD;
                }
            }
        }
    }
    cout<<cnt[n][m]; return 0;
}
```

### 练习 5（进阶）：道路修建最短时间

n×m地图，每个格子有通行时间（1~9分钟），从(1,1)到(n,m)，求最少总时间。

**输入格式：** n m 及时间矩阵（1≤n,m≤20，时间1~9）

**输出格式：** 最少总时间

**样例输入：**
```
3 3
1 3 1
2 1 9
5 1 1
```
**样例输出：** `7`

**思路分析：**

> 有权图！BFS 不再适用（步数不等于时间）。改用**Dijkstra 算法**，或简化为 BFS 分层（优先队列版本）。

**题解代码（Dijkstra 简化版，优先队列）：**

```cpp
#include <bits/stdc++.h>
using namespace std;
int n,m,a[25][25],dis[25][25];
int dx[]={-1,1,0,0}, dy[]={0,0,-1,1};
int main() {
    cin>>n>>m;
    for(int i=1;i<=n;i++) for(int j=1;j<=m;j++) cin>>a[i][j];
    memset(dis,0x3f,sizeof(dis));
    dis[1][1]=a[1][1];
    priority_queue<tuple<int,int,int>, vector<tuple<int,int,int>>, greater<>> pq;
    pq.push({a[1][1],1,1});
    while(!pq.empty()) {
        auto [d,x,y]=pq.top(); pq.pop();
        if(d>dis[x][y]) continue;
        for(int i=0;i<4;i++) {
            int nx=x+dx[i], ny=y+dy[i];
            if(nx>=1&&nx<=n&&ny>=1&&ny<=m&&d+a[nx][ny]<dis[nx][ny]) {
                dis[nx][ny]=d+a[nx][ny];
                pq.push({dis[nx][ny],nx,ny});
            }
        }
    }
    cout<<dis[n][m]; return 0;
}
```

#### 作业 3：消防员最短响应时间

消防局在(1,1)，有多个火灾点（坐标给出），BFS求到每个火灾点的最短步数，输出最大值（最慢响应时间）。

```cpp
#include <bits/stdc++.h>
using namespace std;
int n,m,k,dis[105][105],a[105][105];
int dx[]={-1,1,0,0}, dy[]={0,0,-1,1};
int main() {
    cin>>n>>m>>k;
    for(int i=1;i<=n;i++) for(int j=1;j<=m;j++) cin>>a[i][j];
    memset(dis,-1,sizeof(dis));
    dis[1][1]=0; queue<pair<int,int>> q; q.push({1,1});
    while(!q.empty()) {
        auto [x,y]=q.front(); q.pop();
        for(int i=0;i<4;i++) {
            int nx=x+dx[i], ny=y+dy[i];
            if(nx>=1&&nx<=n&&ny>=1&&ny<=m&&a[nx][ny]==0&&dis[nx][ny]==-1) {
                dis[nx][ny]=dis[x][y]+1; q.push({nx,ny});
            }
        }

    int ans=0;
```

#### 挑战 4：K 步内可达格子数
